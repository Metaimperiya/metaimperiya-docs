🤝 Система друзей

Обзор

Система друзей в METAIMPERIYA работает по принципу "запрос → подтверждение → друзья". Весь процесс построен на атомарных обновлениях Firebase, чтобы статус всегда был синхронизирован у обоих пользователей.

📌 Ключевые принципы:
  • Атомарность — оба статуса обновляются одновременно
  • Реальное время — изменения видны мгновенно через on('value')
  • Единая кнопка — один клик управляет всеми состояниями

Статусы дружбы

У каждого пользователя по отношению к другому может быть один из трёх статусов:

📊 Статусы дружбы 
// ================================================================
// СТАТУСЫ ДРУЖБЫ
// ================================================================

// 1. none — нет связи
//    Кнопка: "➕ Добавить в друзья"

// 2. pending — запрос отправлен (ожидает подтверждения)
//    Кнопка: "⏳ Запрос отправлен"

// 3. friend — в друзьях
//    Кнопка: "✓ В друзьях"

Отправка запроса

При нажатии на кнопку "Добавить" создаётся запись pending в Firebase для обоих пользователей одновременно.

📤 Отправка запроса 
// ================================================================
// ОТПРАВКА ЗАПРОСА В ДРУЗЬЯ
// ================================================================

function sendFriendRequest(targetUid) {
    if (!USER_UID) {
        alert('Войдите!');
        return;
    }
    
    // Проверяем текущий статус
    db.ref('sites/' + SITE + '/friends/' + USER_UID + '/' + targetUid)
        .once('value', snap => {
            const status = snap.val();
            
            if (status === 'friend') {
                alert('✅ Уже в друзьях');
                return;
            }
            
            if (status === 'pending') {
                // Отменяем запрос
                updateFriendStatus(USER_UID, targetUid, null);
                return;
            }
            
            // Отправляем запрос — атомарное создание pending
            updateFriendStatus(USER_UID, targetUid, 'pending');
            
            // Уведомление
            sendNotification(targetUid, {
                type: 'friend_request',
                from: USER_UID,
                text: `${USER} отправил(а) запрос в друзья`,
                timestamp: Date.now()
            });
        });
}

Обработка запроса

Когда пользователь получает запрос, он может принять или отклонить его. Принятие меняет статус на friend.

✅ Принятие запроса 
// ================================================================
// ПРИНЯТИЕ ЗАПРОСА
// ================================================================

function acceptFriendRequest(targetUid) {
    if (!USER_UID) {
        alert('Войдите!');
        return;
    }
    
    // Атомарное обновление
    updateFriendStatus(USER_UID, targetUid, 'friend');
    
    // Уведомление
    sendNotification(targetUid, {
        type: 'friend_accepted',
        from: USER_UID,
        text: `${USER} принял(а) ваш запрос в друзья`,
        timestamp: Date.now()
    });
    
    alert('✅ Дружба подтверждена!');
}

// ================================================================
// ОТКЛОНЕНИЕ ЗАПРОСА
// ================================================================

function declineFriendRequest(targetUid) {
    if (!USER_UID) {
        alert('Войдите!');
        return;
    }
    
    // Удаляем только запрос, подписка остаётся
    const updates = {};
    updates['sites/' + SITE + '/friends/' + USER_UID + '/' + targetUid] = null;
    updates['sites/' + SITE + '/friends/' + targetUid + '/' + USER_UID] = null;
    db.ref().update(updates);
    
    alert('❌ Запрос отклонен');
}

Удаление из друзей

При удалении статус friend удаляется у обоих пользователей.

🗑 Удаление из друзей 
// ================================================================
// УДАЛЕНИЕ ИЗ ДРУЗЕЙ
// ================================================================

function removeFriend(targetUid) {
    if (!USER_UID) {
        alert('Войдите!');
        return;
    }
    
    if (!confirm('Удалить из друзей?')) return;
    
    // Атомарное удаление
    updateFriendStatus(USER_UID, targetUid, null);
    
    alert('❌ Пользователь удален из друзей');
}

Атомарное обновление

Это главная функция, которая гарантирует синхронизацию статусов у обоих пользователей.

⚡ Атомарное обновление статуса 
// ================================================================
// АТОМАРНОЕ ОБНОВЛЕНИЕ СТАТУСА
// ================================================================

function updateFriendStatus(myUid, targetUid, status) {
    if (!myUid || !targetUid) {
        console.error('❌ Ошибка: отсутствуют UID');
        return;
    }
    
    const updates = {};
    const path1 = 'sites/' + SITE + '/friends/' + myUid + '/' + targetUid;
    const path2 = 'sites/' + SITE + '/friends/' + targetUid + '/' + myUid;
    
    if (status === 'friend') {
        updates[path1] = 'friend';
        updates[path2] = 'friend';
        updates['sites/' + SITE + '/followers/' + targetUid + '/' + myUid] = true;
    } else if (status === 'pending') {
        updates[path1] = 'pending';
        updates[path2] = 'pending';
        updates['sites/' + SITE + '/followers/' + targetUid + '/' + myUid] = true;
    } else if (status === null) {
        updates[path1] = null;
        updates[path2] = null;
        updates['sites/' + SITE + '/followers/' + targetUid + '/' + myUid] = null;
    }
    
    // АТОМАРНОЕ ОБНОВЛЕНИЕ — оба пути одновременно
    db.ref().update(updates);
}

// СТРУКТУРА ДРУЗЕЙ В FIREBASE
// ================================================================

// friends — статусы дружбы
sites/metaimperiya/friends/{
    uid_1/{
        uid_2: "friend",      // в друзьях
        uid_3: "pending"     // запрос отправлен
    },
    uid_2/{
        uid_1: "friend"       // в друзьях (синхронизировано)
    }
}

// followers — подписчики
sites/metaimperiya/followers/{
    uid_1/{
        uid_2: true,         // uid_2 подписан на uid_1
        uid_3: true
    }
}
