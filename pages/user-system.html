👤 Система пользователей

Регистрация

В METAIMPERIYA используется два способа регистрации:

  • Google — через Firebase Auth (основной способ)
  • Анонимный — просто по имени (для тестов)
🔐 Регистрация через Google 
// ================================================================
// РЕГИСТРАЦИЯ ЧЕРЕЗ GOOGLE
// ================================================================

// Firebase Auth
const auth = firebase.auth();
const provider = new firebase.auth.GoogleAuthProvider();

// Кнопка "Войти через Google"
document.getElementById('googleBtn').addEventListener('click', () => {
    auth.signInWithPopup(provider).catch(err => {
        if (err.code === 'auth/popup-blocked') {
            auth.signInWithRedirect(provider);
        } else {
            alert('Ошибка входа: ' + err.message);
        }
    });
});

// Слушатель состояния
auth.onAuthStateChanged(user => {
    if (user) {
        // Пользователь успешно зарегистрировался
        USER_UID = user.uid;
        USER = user.displayName || user.email || user.uid.slice(0, 8);
        // Сохраняем в базу
        saveUserData(USER_UID, USER, user.email);
    }
});
👤 Анонимная регистрация 
// ================================================================
// АНОНИМНАЯ РЕГИСТРАЦИЯ
// ================================================================

function loginName() {
    const name = document.getElementById('nameInput').value.trim();
    if (!name) {
        alert('Введите имя');
        return;
    }
    
    auth.signInAnonymously().then(() => {
        USER = name;
        USER_UID = 'anon_' + Date.now();
        localStorage.setItem('dc_u_' + SITE, USER);
        
        // Сохраняем в базу
        const data = {
            name: USER,
            email: 'anon_' + Date.now() + '@anon.com',
            uid: USER_UID,
            lastLogin: Date.now()
        };
        db.ref('sites/' + SITE + '/users/' + USER_UID).update(data);
        db.ref('sites/' + SITE + '/all_users/' + USER_UID).set(data);
    });
}

📋 Профиль

Профиль пользователя содержит:

  • Имя — отображается в списке и на странице
  • Email — для связи
  • Bio — информация о себе
  • Аватарка — загружается в Firebase Storage
  • Username — уникальный идентификатор для ссылки
📄 Сохранение профиля 
// ================================================================
// СОХРАНЕНИЕ ПРОФИЛЯ
// ================================================================

function saveProfile() {
    const name = document.getElementById('editName').value.trim();
    const bio = document.getElementById('editBio').value.trim();
    
    if (!name) {
        alert('Введите имя');
        return;
    }
    
    USER = name;
    localStorage.setItem('dc_u_' + SITE, USER);
    
    // Сохраняем в базу
    const updates = {
        name: USER,
        bio: bio
    };
    db.ref('sites/' + SITE + '/users/' + USER_UID).update(updates);
    db.ref('sites/' + SITE + '/all_users/' + USER_UID).update(updates);
}

📷 Аватарка

Аватарки загружаются в Firebase Storage и сохраняются в профиле.

📷 Загрузка аватарки 
// ================================================================
// ЗАГРУЗКА АВАТАРКИ
// ================================================================

function uploadAvatar() {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'image/*';
    
    input.onchange = function(e) {
        const file = e.target.files[0];
        if (!file) return;
        
        // Проверка размера (максимум 5 МБ)
        if (file.size > 5 * 1024 * 1024) {
            alert('Максимум 5 МБ');
            return;
        }
        
        // Путь в Storage: avatars/{uid}/{timestamp}_{filename}
        const ref = storage.ref(
            'avatars/' + USER_UID + '/' + Date.now() + '_' + file.name
        );
        
        ref.put(file)
            .then(snap => snap.ref.getDownloadURL())
            .then(url => {
                // Сохраняем ссылку в базу
                db.ref('sites/' + SITE + '/users/' + USER_UID + '/avatarUrl').set(url);
                db.ref('sites/' + SITE + '/all_users/' + USER_UID + '/avatarUrl').set(url);
                
                // Обновляем UI
                updateUI();
                loadProfile();
                alert('✅ Аватарка обновлена!');
            });
    };
    
    input.click();
}

🔗 Username

Каждый пользователь имеет уникальный username, который используется в ссылках:

  • Формат: транслитерация имени + нижний регистр
  • Пример: playerlikee → metaimperiya.com/playerlikee
  • Хранение: в Firebase в узле users_by_name
🔗 Генерация username 
// ================================================================
// ГЕНЕРАЦИЯ USERNAME
// ================================================================

function generateUsername(name) {
    // Транслитерация кириллицы
    const translitMap = {
        'а':'a', 'б':'b', 'в':'v', 'г':'g',
        'д':'d', 'е':'e', 'ё':'e', 'ж':'zh',
        'з':'z', 'и':'i', 'й':'y', 'к':'k',
        'л':'l', 'м':'m', 'н':'n', 'о':'o',
        'п':'p', 'р':'r', 'с':'s', 'т':'t',
        'у':'u', 'ф':'f', 'х':'h', 'ц':'ts',
        'ч':'ch', 'ш':'sh', 'щ':'sch',
        'ъ':'', 'ы':'y', 'ь':'',
        'э':'e', 'ю':'yu', 'я':'ya'
    };
    
    let username = '';
    for (let char of name.trim()) {
        username += translitMap[char] || char;
    }
    
    // Убираем спецсимволы
    username = username.replace(/[^a-zA-Z0-9_]/g, '').toLowerCase();
    
    return username;
}

// Сохранение username в базу
function saveUsername(uid, username) {
    db.ref('sites/' + SITE + '/users_by_name/' + username).set(uid);
    db.ref('sites/' + SITE + '/users/' + uid + '/username').set(username);
    db.ref('sites/' + SITE + '/all_users/' + uid + '/username').set(username);
}
