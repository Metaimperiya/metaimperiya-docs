💬 Система чатов

Обзор

В METAIMPERIYA реализованы личные сообщения между пользователями. Каждый диалог — это отдельная комната с уникальным ID, который одинаков для обоих собеседников.

📌 Ключевые принципы:
  • Уникальный ID — генерируется из UID обоих пользователей
  • Реальное время — сообщения приходят мгновенно через on('child_added')
  • Уведомления — приходит оповещение о новом сообщении
  • Fixed-окно — чат открывается поверх всего интерфейса

Генерация ID чата

ID чата создаётся путём сортировки UID двух пользователей:

🔑 Генерация ID чата 
// ================================================================
// ГЕНЕРАЦИЯ УНИКАЛЬНОГО ID ЧАТА
// ================================================================

// UID сортируются, чтобы у обоих собеседников был одинаковый ID
const chatId = [USER_UID, targetUid].sort().join('_');

// Пример
// USER_UID = "uid_abc"
// targetUid = "uid_xyz"
// chatId = "uid_abc_uid_xyz"

// Путь в Firebase
const path = 'dms/' + SITE + '/' + chatId + '/messages';

Открытие чата

Функция открывает окно чата и подгружает историю сообщений.

📂 Открытие чата 
// ================================================================
// ОТКРЫТИЕ ЧАТА
// ================================================================

function openPrivateChat(targetUid) {
    if (!USER_UID) {
        alert('Войдите!');
        return;
    }
    
    if (targetUid === USER_UID) {
        alert('Нельзя писать себе');
        return;
    }
    
    // Генерируем ID чата
    const chatId = [USER_UID, targetUid].sort().join('_');
    const path = 'dms/' + SITE + '/' + chatId + '/messages';
    
    // Закрываем все модалки
    closeNotifications();
    closeProfileView();
    closeContextMenu();
    
    CURRENT_ROOM = chatId;
    
    // Показываем окно чата
    document.getElementById('chatWindow').classList.add('active');
    
    // Загружаем имя собеседника
    db.ref('sites/' + SITE + '/all_users/' + targetUid)
        .once('value', snap => {
            const user = snap.val() || {};
            const name = user.name || 'Собеседник';
            document.getElementById('chatTitle').textContent = '💬 ' + name;
        });
    
    loadChat(path);
}

Отправка сообщения

Сообщение сохраняется в Firebase и отправляется получателю.

📤 Отправка сообщения 
// ================================================================
// ОТПРАВКА СООБЩЕНИЯ
// ================================================================

function sendChat() {
    const input = document.getElementById('chatInput');
    const text = input.value.trim();
    
    if (!text || !CURRENT_ROOM) return;
    
    const path = 'dms/' + SITE + '/' + CURRENT_ROOM + '/messages';
    
    // Определяем собеседника
    const parts = CURRENT_ROOM.split('_');
    const targetUid = parts[0] === USER_UID ? parts[1] : parts[0];
    
    // Сохраняем сообщение
    db.ref(path).push({
        nick: USER,
        text: text,
        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        timestamp: Date.now()
    });
    
    // Отправляем уведомление собеседнику
    sendNotification(targetUid, {
        type: 'message',
        from: USER_UID,
        text: `${USER}: ${text}`,
        timestamp: Date.now()
    });
    
    input.value = '';
}

Загрузка сообщений

Сообщения загружаются в реальном времени через слушатель on('value').

📥 Загрузка сообщений 
// ================================================================
// ЗАГРУЗКА СООБЩЕНИЙ В РЕАЛЬНОМ ВРЕМЕНИ
// ================================================================

function loadChat(path) {
    // Отписываемся от старого слушателя
    if (chatUnsub) {
        if (typeof chatUnsub === 'string') {
            db.ref(chatUnsub).off('value');
        }
        chatUnsub = null;
    }
    
    const box = document.getElementById('chatMessages');
    box.innerHTML = '<div style="color:#65676b;text-align:center;padding:20px;">⏳ Загрузка...</div>';
    
    chatUnsub = path;
    
    // Слушатель изменений
    db.ref(path)
        .orderByChild('timestamp')
        .on('value', snap => {
            box.innerHTML = '';
            
            if (!snap.exists()) {
                box.innerHTML = '<div style="color:#65676b;text-align:center;padding:20px;">Сообщений нет</div>';
                return;
            }
            
            snap.forEach(s => {
                const m = s.val();
                const isSelf = m.nick === USER;
                const div = document.createElement('div');
                div.className = 'msg' + (isSelf ? ' self' : '');
                div.innerHTML = `
                    <span class="author">${esc(m.nick||'?')}</span>
                    <span class="time">${m.time||''}</span>
                    <div>${esc(m.text||'')}</div>
                `;
                box.appendChild(div);
            });
            
            box.scrollTop = box.scrollHeight;
        });
}

Уведомления о сообщениях

При отправке сообщения собеседнику приходит уведомление.

🔔 Уведомления 
// ================================================================
// УВЕДОМЛЕНИЯ
// ================================================================

function sendNotification(targetUid, data) {
    if (!USER_UID) return;
    if (!targetUid || targetUid === USER_UID) {
        console.warn('❌ Нельзя отправить уведомление себе');
        return;
    }
    
    const notifRef = db.ref('sites/' + SITE + '/notifications/' + targetUid).push();
    notifRef.set({
        ...data,
        read: false,
        fromName: USER,
        fromUid: USER_UID
    });
}

// Обработка клика по уведомлению
function handleNotifClick(notifId, fromUid, type) {
    if (type === 'message') {
        closeNotifications();
        openPrivateChat(fromUid);
        // Удаляем уведомление
        db.ref('sites/' + SITE + '/notifications/' + USER_UID + '/' + notifId).remove();
    }
}
