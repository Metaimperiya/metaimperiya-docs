👑 Админ-панель

Обзор

Админ-панель в METAIMPERIYA предназначена для управления пользователями и контентом. Доступ к ней открывается автоматически при входе с аккаунта, чей UID есть в списке администраторов.

📌 Ключевые принципы:
  • Автоматическое распознавание — админка включается по UID
  • Безопасность — все функции проверяют права
  • Простота — управление в пару кликов

UID Админов

Список администраторов жёстко закодирован в массиве. Если ты хочешь добавить нового админа — просто добавь его UID в этот список.

👑 Список администраторов 
// ================================================================
// АДМИНЫ — ХАРДКОД UID
// ================================================================

const ADMIN_UIDS = [
    "ayXehcol9FgAQU6tZuup7aSaRoV2",  // Все Работники
    "pWB0nGVvVXc4je6466ss7IwBm9G2",  // Джобс Москва
    "ANR62p3qcjOe2ALsdVvJHUNCCV42"   // Плеер Лайки
];

let isAdmin = false;

Проверка доступа

Функция проверяет, есть ли текущий пользователь в списке администраторов.

🔐 Проверка админ-доступа 
// ================================================================
// ПРОВЕРКА АДМИН-ДОСТУПА
// ================================================================

function checkAdminAccess(uid) {
    const wasAdmin = isAdmin;
    isAdmin = ADMIN_UIDS.includes(uid);
    
    if (wasAdmin !== isAdmin) {
        updateAdminUI();
    }
    
    console.log(`👑 Админ-режим: ${isAdmin ? 'ВКЛЮЧЕН' : 'ВЫКЛЮЧЕН'}`);
    return isAdmin;
}

// Вызывается при входе
auth.onAuthStateChanged(user => {
    if (user) {
        USER_UID = user.uid;
        checkAdminAccess(USER_UID);
        // ...
    }
});

Удаление пользователя

Администратор может удалить любого пользователя. Это действие необратимо.

🗑 Удаление пользователя 
// ================================================================
// УДАЛЕНИЕ ПОЛЬЗОВАТЕЛЯ
// ================================================================

function adminDeleteUser(uid) {
    if (!isAdmin) {
        console.warn('❌ Доступ запрещен');
        return;
    }
    
    if (!uid || uid === USER_UID) {
        alert('Нельзя удалить себя');
        return;
    }
    
    if (!confirm('Удалить пользователя? Это необратимо!')) return;
    
    const updates = {};
    updates['sites/' + SITE + '/users/' + uid] = null;
    updates['sites/' + SITE + '/all_users/' + uid] = null;
    updates['sites/' + SITE + '/friends/' + uid] = null;
    updates['sites/' + SITE + '/followers/' + uid] = null;
    updates['sites/' + SITE + '/notifications/' + uid] = null;
    updates['sites/' + SITE + '/users_by_name/'] = null;
    
    db.ref().update(updates)
        .then(() => {
            alert('✅ Пользователь удален');
            loadPeople();
        })
        .catch(err => {
            console.error('❌ Ошибка:', err);
            alert('Ошибка удаления');
        });
}

Очистка данных

Администратор может очистить все комнаты или все уведомления.

🧹 Очистка данных 
// ================================================================
// ОЧИСТКА КОМНАТ
// ================================================================

function adminClearRooms() {
    if (!isAdmin) {
        console.warn('❌ Доступ запрещен');
        return;
    }
    
    if (!confirm('Очистить все комнаты?')) return;
    
    db.ref('sites/' + SITE + '/rooms').remove()
        .then(() => {
            db.ref('sites/' + SITE + '/room_users/').remove();
            alert('✅ Комнаты очищены');
            loadRooms();
        })
        .catch(err => {
            console.error('❌ Ошибка:', err);
            alert('Ошибка очистки');
        });
}

// ================================================================
// ОЧИСТКА УВЕДОМЛЕНИЙ
// ================================================================

function adminClearNotifications() {
    if (!isAdmin) {

    db.ref('sites/' + SITE + '/notifications/').remove()

        })
        .catch(err => {
            console.error('❌ Ошибка:', err);
            alert('Ошибка очистки');
        });
}

Управление username

Администратор может вручную устанавливать и удалять username для любого пользователя.

✏️ Установка username 
// ================================================================
// УСТАНОВКА USERNAME
// ================================================================

function adminSetUsername(uid, customUsername) {
    if (!isAdmin) {
        console.warn('❌ Доступ запрещен');
        return;
    }
    
    if (!uid || !customUsername) {
        alert('❌ Нужно указать UID и username');
        return;
    }
    
    // Проверяем, что username не занят
    db.ref('sites/' + SITE + '/users_by_name/' + customUsername)
        .once('value', snap => {
            if (snap.val() && snap.val() !== uid) {
                alert(`❌ Username "${customUsername}" уже занят!`);
                return;
            }
            
            const updates = {};
            updates['sites/' + SITE + '/users_by_name/' + customUsername] = uid;
            updates['sites/' + SITE + '/users/' + uid + '/username'] = customUsername;
            updates['sites/' + SITE + '/all_users/' + uid + '/username'] = customUsername;
            
            db.ref().update(updates)
                .then(() => {
                    alert(`✅ Username "${customUsername}" привязан!`);
                    loadPeople();
                })
                .catch(err => {
                    console.error('❌ Ошибка:', err);
                    alert('Ошибка сохранения');
                });
        });
}
