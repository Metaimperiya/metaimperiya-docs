🔍 SEO-оптимизация

Обзор

METAIMPERIYA использует несколько подходов для SEO-оптимизации, чтобы страницы хорошо индексировались поисковыми системами.

📌 Ключевые принципы:
  • Динамические мета-теги — генерируются для каждой страницы
  • Open Graph — для соцсетей
  • Статические URL — каждый профиль — отдельная страница
  • Sitemap — карта сайта для Google

Мета-теги

В проекте используются стандартные мета-теги для SEO.

📄 Базовые мета-теги 
<!-- ===== SEO META ТЕГИ ===== -->
<title id="seoTitle">METAIMPERIYA - Социальная сеть</title>
<meta name="description" id="seoDescription" content="Социальная сеть нового поколения">
<meta name="keywords" id="seoKeywords" content="социальная сеть, общение, друзья">

<!-- Open Graph для соцсетей -->
<meta property="og:title" id="ogTitle" content="METAIMPERIYA">
<meta property="og:description" id="ogDescription" content="Социальная сеть нового поколения">
<meta property="og:type" content="website">
<meta property="og:url" id="ogUrl" content="https://metaimperiya.com">
<meta property="og:image" id="ogImage" content="https://metaimperiya.com/og-image.jpg">

<!-- Canonical URL -->
<link rel="canonical" id="canonicalUrl" href="https://metaimperiya.com/">

<!-- Robots -->
<meta name="robots" id="robotsMeta" content="index, follow">

Динамическая генерация мета-тегов

Функция generateMeta(data) обновляет мета-теги на основе данных из Firebase.

⚡ Генерация мета-тегов 
// ================================================================
// ГЕНЕРАЦИЯ МЕТА-ТЕГОВ
// ================================================================

function generateMeta(data) {
    if (!data) {
        // Дефолтные мета-теги
        document.getElementById('seoTitle').textContent = 'METAIMPERIYA - Социальная сеть';
        document.getElementById('seoDescription').textContent = 'Социальная сеть нового поколения';
        document.getElementById('seoKeywords').textContent = 'социальная сеть, общение, друзья';
        return;
    }
    
    // Устанавливаем мета-теги из данных
    if (data.title) {
        document.getElementById('seoTitle').textContent = data.title + ' | METAIMPERIYA';
        document.getElementById('ogTitle').content = data.title + ' | METAIMPERIYA';
        document.title = data.title + ' | METAIMPERIYA';
    }
    
    if (data.description) {
        document.getElementById('seoDescription').textContent = data.description;
        document.getElementById('ogDescription').content = data.description;
    }
    
    if (data.keywords) {
        document.getElementById('seoKeywords').textContent = data.keywords;
    }
}

SEO-страницы

Для каждого пользователя и группы можно создать отдельную SEO-страницу с уникальными мета-тегами.

📄 Загрузка SEO-страницы 
// ================================================================
// ЗАГРУЗКА SEO-СТРАНИЦЫ
// ================================================================

function loadSEOPage(path) {
    const slug = path.startsWith('/') ? path.substring(1) : path;
    if (!slug) return false;
    
    // Проверяем пользователя
    db.ref('sites/' + SITE + '/users_by_name/' + slug).once('value', snap => {
        const uid = snap.val();
        if (uid) {
            db.ref('sites/' + SITE + '/all_users/' + uid).once('value', userSnap => {
                const user = userSnap.val();
                if (user) {
                    const seoData = {
                        title: user.name || slug,
                        description: user.bio || `Профиль пользователя ${user.name || slug}`,
                        keywords: `${user.name || slug}, профиль, пользователь`,
                        image: user.avatarUrl || 'https://metaimperiya.com/og-image.jpg'
                    };
                    generateMeta(seoData);
                }
            });
        }
    });
}

🗺️ Sitemap

Генерация Sitemap для Google. Собирает все страницы: пользователей, группы и SEO-страницы.

🗺️ Генерация Sitemap 
// ================================================================
// ГЕНЕРАЦИЯ SITEMAP
// ================================================================

function adminGenerateSitemap() {
    if (!isAdmin) {
        console.warn('❌ Доступ запрещен');
        return;
    }
    
    const paths = [];
    
    // Собираем пользователей
    db.ref('sites/' + SITE + '/all_users').once('value', snap => {
        const users = snap.val() || {};
        for (const uid in users) {
            if (users[uid].username) {
                paths.push(users[uid].username);
            }
        }
        
        // Собираем SEO-страницы
        db.ref('sites/' + SITE + '/pages_seo').once('value', snap2 => {
            const pages = snap2.val() || {};
            for (const key in pages) {
                if (!paths.includes(key)) {
                    paths.push(key);
                }
            }
            
            // Генерируем XML
            const baseUrl = window.location.origin;
            let xml = '<?xml version="1.0" encoding="UTF-8"?>\n';
            xml += '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n';
            xml += `  <url>\n    <loc>${baseUrl}/</loc>\n    <priority>1.0</priority>\n  </url>\n`;
            
            paths.forEach(path => {
                if (path && path.trim() !== '') {
                    xml += `  <url>\n    <loc>${baseUrl}/${path}</loc>\n    <priority>0.8</priority>\n  </url>\n`;
                }
            });
            
            xml += '</urlset>';
            
            // Скачиваем
            const blob = new Blob([xml], { type: 'application/xml' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'sitemap.xml';
            a.click();
            URL.revokeObjectURL(url);
            
            alert(`✅ Sitemap сгенерирован! ${paths.length} страниц`);
        });
    });
}

    // Позволяет задать Title, Description, Keywords

function adminSaveSEO() {
    const path = document.getElementById('seoPath').value.trim();
    const title = document.getElementById('seoTitleInput').value.trim();
    const description = document.getElementById('seoDescriptionInput').value.trim();
    const keywords = document.getElementById('seoKeywordsInput').value.trim();
    
    if (!path || !title) {
        alert('❌ Введите путь и заголовок');
        return;
    }
    
    const data = {
        title: title,
        description: description || title,
        keywords: keywords || title,
        updatedAt: Date.now()
    };
    
    db.ref('sites/' + SITE + '/pages_seo/' + path).set(data)
        .then(() => {
            alert('✅ SEO-данные сохранены!');
        })
        .catch(err => {
            console.error('❌ Ошибка:', err);
            alert('Ошибка сохранения');
        });
}
