🔐 Аутентификация

В проекте используется два способа входа:

  • Google — через Firebase Auth
  • Анонимный — без пароля и email
🔐 Код авторизации 
// ===== AUTH =====
// Слушатель состояния
auth.onAuthStateChanged(user => {
    if (user) {
        // Пользователь вошел
        USER_UID = user.uid;
        USER = user.displayName || user.email || user.uid.slice(0, 8);
        // ... обновляем UI
    } else {
        // Показываем окно входа
        document.getElementById('loginModal').classList.add('open');
    }
});

// Вход через Google
document.getElementById('googleBtn').addEventListener('click', () => {
    auth.signInWithPopup(provider);
});

// Анонимный вход
function loginName() {
    const name = document.getElementById('nameInput').value.trim();
    if (!name) { alert('Введите имя'); return; }
    auth.signInAnonymously();
}

📷 Хранилище (Storage)

Аватарки пользователей хранятся в Firebase Storage:

  • Путь: avatars/{uid}/{timestamp}_{filename}
  • Максимальный размер: 5 MB
  • Форматы: JPG, PNG, GIF
📷 Загрузка аватарки 
function uploadAvatar() {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'image/*';
    input.onchange = function(e) {
        const file = e.target.files[0];
        if (!file) return;
        if (file.size > 5 * 1024 * 1024) { alert('Максимум 5 МБ'); return; }
        
        const ref = storage.ref('avatars/' + USER_UID + '/' + Date.now() + '_' + file.name);
        ref.put(file).then(snap => snap.ref.getDownloadURL()).then(url => {
            // Сохраняем URL в базу
            db.ref('sites/' + SITE + '/users/' + USER_UID + '/avatarUrl').set(url);
            db.ref('sites/' + SITE + '/all_users/' + USER_UID + '/avatarUrl').set(url);
        });
    };
    input.click();
}
